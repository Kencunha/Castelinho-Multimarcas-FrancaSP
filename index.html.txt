function runCounter(el) {
  if (el.dataset.done) return;
  el.dataset.done = '1';
  const target = parseFloat(el.dataset.target);
  const dec = parseInt(el.dataset.decimals || '0');
  const dur = 1400, start = performance.now();
  const step = now => {
    const p = Math.min((now - start) / dur, 1);
    const val = target * (1 - Math.pow(1 - p, 3));
    el.textContent = val.toLocaleString('pt-BR', { minimumFractionDigits: dec, maximumFractionDigits: dec });
    if (p < 1) requestAnimationFrame(step);
  };
  requestAnimationFrame(step);
}

function buildCar(color) {
  const car = new THREE.Group();
  const bodyMat = new THREE.MeshStandardMaterial({ color, metalness: 0.7, roughness: 0.22 });
  const glassMat = new THREE.MeshStandardMaterial({ color: 0x0b1620, metalness: 0.9, roughness: 0.05 });
  const blackMat = new THREE.MeshStandardMaterial({ color: 0x0a0a0a, roughness: 0.7 });
  const chromeMat = new THREE.MeshStandardMaterial({ color: 0xcfd6dd, metalness: 1, roughness: 0.15 });
  const lightMat = new THREE.MeshStandardMaterial({ color: 0xffffee, emissive: 0xffffcc, emissiveIntensity: 1.2 });
  const tailMat = new THREE.MeshStandardMaterial({ color: 0xff0000, emissive: 0xff0000, emissiveIntensity: 1 });

  const add = (geo, mat, x, y, z, rot = 0) => {
    const m = new THREE.Mesh(geo, mat);
    m.position.set(x, y, z);
    if (rot) m.rotation.z = rot;
    m.castShadow = true;
    m.receiveShadow = true;
    car.add(m);
    return m;
  };

  // carroceria inferior
  add(new THREE.BoxGeometry(4.2, 0.85, 1.9), bodyMat, 0, 0.8, 0);
  // capô e porta-malas com leve inclinação
  add(new THREE.BoxGeometry(1.2, 0.2, 1.8), bodyMat, 1.5, 1.2, 0, -0.08);

  // cabine (vidros)
  add(new THREE.BoxGeometry(2.3, 0.85, 1.72), glassMat, -0.1, 1.67, 0);
  // teto
  add(new THREE.BoxGeometry(1.9, 0.12, 1.66), bodyMat, -0.1, 2.15, 0);
  // para-choque e soleira pretos
  add(new THREE.BoxGeometry(0.2, 0.35, 1.85), blackMat, 2.12, 0.6, 0);
  add(new THREE.BoxGeometry(0.2, 0.35, 1.85), blackMat, -2.12, 0.6, 0);
  // grade e faróis
  add(new THREE.BoxGeometry(0.06, 0.35, 1.2), chromeMat, 2.12, 0.95, 0);
  add(new THREE.BoxGeometry(0.06, 0.2, 0.4), lightMat, 2.12, 1.1, 0.62);
  add(new THREE.BoxGeometry(0.06, 0.2, 0.4), lightMat, 2.12, 1.1, -0.62);

  add(new THREE.BoxGeometry(0.06, 0.2, 0.4), tailMat, -2.12, 1.1, 0.62);
  add(new THREE.BoxGeometry(0.06, 0.2, 0.4), tailMat, -2.12, 1.1, -0.62);

  const wheelGeo = new THREE.CylinderGeometry(0.42, 0.42, 0.34, 28);

  [[1.4, 0.42, 0.98], [1.4, 0.42, -0.98], [-1.4, 0.42, 0.98], [-1.4, 0.42, -0.98]].forEach(([x, y, z]) => {

    r.rotation.x = Math.PI / 2;
    r.position.set(x, y, z);
    car.add(r);
  });

  car.userData.bodyMat = bodyMat;

}

  const i = +sTax.value / 100;
  const parcela = i === 0 ? financiado / n : financiado * i / (1 - Math.pow(1 + i, -n));
  const total = parcela * n;

  $('#vValor').textContent = fmt(valor);
  $('#vEntrada').textContent = `${entradaPct}% (${fmt(entrada)})`;
  $('#vPrazo').textContent = `${n} meses`;
  $('#vTaxa').textContent = `${(+sTax.value).toFixed(2).replace('.', ',')}% a.m.`;
  $('#resParcela').textContent = `${fmt(parcela)}/mês`;
  $('#resDetalhes').innerHTML = `Valor financiado: <strong>${fmt(financiado)}</strong><br>Total pago: <strong>${fmt(total)}</strong>`;
  $('#resBtn').onclick = e => {
    e.preventDefault();
    const txt = encodeURIComponent(`Olá! Simulei: veículo ${fmt(valor)}, entrada ${entradaPct}%, ${n}x de ${fmt(parcela)}. Quero saber mais.`);
    window.open(`https://wa.me/${WHATS}?text=${txt}`, '_blank');
  };
}

  slides[slideIdx].classList.remove('active');
  dotsWrap.children[slideIdx].classList.remove('active');
  slideIdx = i;
  slides[slideIdx].classList.add('active');
  dotsWrap.children[slideIdx].classList.add('active');
}
setInterval(() => goTo((slideIdx + 1) % slides.length), 5000);
